cmake_minimum_required(VERSION 3.24)
project(jarjar_combat LANGUAGES CXX)

set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
set(GODOTCPP_TARGET "template_release" CACHE STRING "" FORCE)
set(GODOTCPP_PRECISION "single" CACHE STRING "" FORCE)
set(GODOTCPP_BUILD_PROFILE "${CMAKE_CURRENT_SOURCE_DIR}/build_profile.json" CACHE FILEPATH "" FORCE)
add_subdirectory("${COMBAT_SDK_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/godot-cpp")
add_library(jarjar_combat SHARED combat_kernel.cpp combat_world.cpp)
target_link_libraries(jarjar_combat PRIVATE godot::cpp)
target_compile_features(jarjar_combat PRIVATE cxx_std_17)
target_compile_options(jarjar_combat PRIVATE /O2 /fp:strict /W4 /utf-8)
set_target_properties(godot-cpp PROPERTIES INTERFACE_SYSTEM_INCLUDE_DIRECTORIES "$<TARGET_PROPERTY:godot-cpp,INTERFACE_INCLUDE_DIRECTORIES>")
set_target_properties(jarjar_combat PROPERTIES RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_CURRENT_BINARY_DIR}/bin")
